SELECT *
FROM Producers
WHERE 
(
    SELECT 
        SUM(CASE WHEN TradeType = 'BUY' THEN Quantity ELSE 0 END) -
        SUM(CASE WHEN TradeType = 'SELL' THEN Quantity ELSE 0 END)
    FROM Trades
    WHERE Producers.ProducerID = ?
    AND Trades.ProducerID = Producers.ProducerID
) <= 0
AND StockBudget <= 0;
